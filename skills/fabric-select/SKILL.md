# Fabric Select

Use one `fabric_exec` call to reduce 2–6 candidates to one winner plus graft suggestions. The ladder, in order: anything external that can check a candidate (tests, compilation, execution) beats every judge — run it first and let it eliminate; judging is pairwise and order-swapped, never absolute scores; risk criteria are per-criterion vetoes where one failed check eliminates regardless of pairwise wins; and the losers are mined for what they do better before being discarded.

Pass `strings.task` (what the candidates were asked to do), `strings.candidates` as JSON `Array<{id, content}>` (or `content` as a file path when candidates live on disk — give judges `read` and reference the path), optional `strings.check` (a bash command template with `{id}` substituted, exit 0 = pass) and `strings.criteria` (JSON array of atomic yes/no risk criteria). Judges compare and score; they never see who or what generated a candidate. If the candidates were generated by this session's model, pass `strings.judgeModel` naming a different model — a generator's own family judging its output favors it.

```ts
type Pair = { a: number; b: number };
type PairVerdict = { winner: "A" | "B" | "tie"; reason: string };
type CriterionVerdict = { pass: boolean; evidence: string };

const candidates = JSON.parse(π.candidates) as Array<{ id: string; content: string }>;
if (candidates.length < 2 || candidates.length > 6) {
  throw new Error("Select requires 2–6 candidates.");
}
const criteria = π.criteria ? (JSON.parse(π.criteria) as string[]) : [];
const judgeOpts = {
  tools: ["read", "grep", "find", "ls"],
  ...(π.judgeModel ? { model: π.judgeModel } : {}),
};

await workflow.configure({
  name: "Candidate selection",
  description: `${candidates.length} candidates → external check → pairwise → veto → graft`,
});

// Rung 1 — external check (only when a real verifier exists; a judge is the fallback, not the default)
let alive = candidates.map((c, i) => ({ ...c, index: i }));
const eliminated: Array<{ id: string; stage: string; reason: string }> = [];
if (π.check) {
  await phase("External check", { total: alive.length });
  const checked = await parallel(
    alive.map((c) => async () => {
      const { ok, output } = await pi.bash({ command: π.check.split("{id}").join(c.id) });
      return { c, ok, output: String(output ?? "").slice(0, 2000) };
    }),
    { concurrency: alive.length },
  );
  for (const r of checked) {
    if (!r.ok) eliminated.push({ id: r.c.id, stage: "external-check", reason: r.output });
  }
  const failedIds = new Set(eliminated.map((e) => e.id));
  alive = alive.filter((c) => !failedIds.has(c.id));
  if (alive.length === 0) return { status: "failed", winner: null, eliminated, reason: "every candidate failed the external check" };
  if (alive.length === 1) return { status: "success", winner: alive[0].id, standings: [], eliminated, grafts: [] };
}

// Rung 2 — pairwise, both orders; disagreement between orders = tie
const pairs: Pair[] = [];
for (let i = 0; i < alive.length; i++) for (let j = i + 1; j < alive.length; j++) pairs.push({ a: i, b: j });
await phase("Pairwise", { total: pairs.length * 2 });
const judgePair = (first: number, second: number) =>
  agent<PairVerdict>(
    `Two anonymous candidate solutions to the same task follow. Judge ONLY against the task requirements — ignore length, style, and confidence of phrasing; verbosity is not quality. First critique each independently, then pick the better one, or "tie" when neither is clearly better.\n\nTask:\n${π.task}\n\nCandidate A:\n${alive[first].content}\n\nCandidate B:\n${alive[second].content}`,
    {
      ...judgeOpts,
      label: `judge ${alive[first].id} vs ${alive[second].id}`.slice(0, 50),
      schema: {
        type: "object",
        properties: { winner: { type: "string", enum: ["A", "B", "tie"] }, reason: { type: "string" } },
        required: ["winner", "reason"],
        additionalProperties: false,
      },
    },
  );
const results = await parallel(
  pairs.flatMap(({ a, b }) => [
    async () => ({ a, b, swapped: false, v: await judgePair(a, b) }),
    async () => ({ a, b, swapped: true, v: await judgePair(b, a) }),
  ]),
  { concurrency: 8 },
);
const wins = new Map<number, number>(alive.map((_, i) => [i, 0]));
for (const { a, b } of pairs) {
  const fwd = results.find((r) => r.a === a && r.b === b && !r.swapped)?.v;
  const rev = results.find((r) => r.a === a && r.b === b && r.swapped)?.v;
  // Normalize the swapped verdict back to (a, b), then require agreement
  const fwdWinner = fwd?.winner === "A" ? a : fwd?.winner === "B" ? b : null;
  const revWinner = rev?.winner === "A" ? b : rev?.winner === "B" ? a : null;
  if (fwdWinner !== null && fwdWinner === revWinner) wins.set(fwdWinner, (wins.get(fwdWinner) ?? 0) + 1);
  // order-dependent disagreement counts as a tie: no points
}
const standings = [...alive]
  .map((c) => ({ id: c.id, index: c.index, wins: wins.get(alive.indexOf(c)) ?? 0 }))
  .sort((x, y) => y.wins - x.wins);

// Rung 3 — per-criterion veto on the leader; one failed risk criterion eliminates
await phase("Veto", { total: criteria.length || 1 });
const vetoes: Array<{ id: string; criterion: string; evidence: string }> = [];
let winner: string | null = null;
for (const s of standings) {
  const cand = alive.find((c) => c.id === s.id)!;
  let vetoed = false;
  for (const criterion of criteria) {
    const verdict = await agent<CriterionVerdict>(
      `Check exactly one criterion against this candidate. Answer pass/fail with verbatim evidence quoted from the candidate; when uncertain, fail.\n\nCriterion: ${criterion}\n\nTask:\n${π.task}\n\nCandidate:\n${cand.content}`,
      {
        ...judgeOpts,
        label: `veto: ${criterion}`.slice(0, 50),
        thinking: "xhigh",
        schema: {
          type: "object",
          properties: { pass: { type: "boolean" }, evidence: { type: "string" } },
          required: ["pass", "evidence"],
          additionalProperties: false,
        },
      },
    );
    if (!verdict.pass) { vetoes.push({ id: cand.id, criterion, evidence: verdict.evidence }); vetoed = true; break; }
  }
  if (!vetoed) { winner = cand.id; break; }
}
if (!winner) return { status: "failed", winner: null, standings, vetoes, eliminated, reason: "every candidate was vetoed" };

// Rung 4 — graft: mine the losers before discarding them
await phase("Graft", { total: 1 });
const losers = alive.filter((c) => c.id !== winner);
const grafts = losers.length === 0 ? [] : await agent<{ grafts: Array<{ from: string; suggestion: string }> }>(
  `Candidate ${winner} won. For each losing candidate, name anything it does concretely better than the winner that is worth splicing into it — or state it contributes nothing. Do not relitigate the ranking.\n\nTask:\n${π.task}\n\nWinner:\n${alive.find((c) => c.id === winner)!.content}\n\nLosers:\n${JSON.stringify(losers)}`,
  {
    ...judgeOpts,
    label: "graft",
    schema: {
      type: "object",
      properties: { grafts: { type: "array", items: { type: "object", properties: { from: { type: "string" }, suggestion: { type: "string" } }, required: ["from", "suggestion"], additionalProperties: false } } },
      required: ["grafts"],
      additionalProperties: false,
    },
  },
).then((r) => r.grafts);

return { status: "success", winner, standings, vetoes, eliminated, grafts };
```

Cost is `2·C(n,2)` pairwise calls plus up to `n × criteria` veto calls plus one graft call — reserve that agent budget before starting; with 6 candidates and 4 criteria this is ~55 calls, so prune with an external check or a cheap screening judge first when candidates are many. Agreement across judges measures confidence, not truth — a unanimous pairwise sweep with a failed veto is still an elimination. `bash` in the external check requires execute approval. Return the winner and grafts to the caller; applying grafts is a separate edit task, not part of selection.
